### Листинг 2. Запись, чтение и вызовы

```text
export function saveTasks(storage, key, tasks) {
  if (!isValidTaskList(tasks)) {
    return { ok: false, error: "Сохранять можно только корректный список задач." };
  }

  try {
    storage.setItem(key, JSON.stringify({ version: STORAGE_VERSION, tasks }));
  } catch (error) {
    return { ok: false, error: `Не удалось сохранить данные: ${error.message}.` };
  }

  return { ok: true };
}

export function loadTasks(storage, key, fallbackTasks) {
  // getItem в try/catch; при отсутствии ключа — исходный набор
  parsed = JSON.parse(raw);

  if (
    parsed === null ||
    typeof parsed !== "object" ||
    parsed.version !== STORAGE_VERSION ||
    !isValidTaskList(parsed.tasks)
  ) {
    return { ok: false, source: "fallback", tasks: copyTasks(fallbackTasks), error: "..." };
  }

  return { ok: true, source: "storage", tasks: copyTasks(parsed.tasks) };
}

function persistTasks(tasks) {
  const result = saveTasks(window.localStorage, storageKey, tasks);
  // storageKey для общего набора: tip-js-practice-04:demo
}

function restoreTasks() {
  const result = loadTasks(window.localStorage, storageKey, initialTasks);
  currentTasks = result.tasks;
}
```

`STORAGE_VERSION = 1`. `persistTasks` вызывается после успешного добавления, правки, смены статуса и удаления. `restoreTasks` вызывается при старте страницы.

## 3. Получение задач с сервера — 1 балл

API ПР5: `npm start` в `practice-05`, адрес `http://127.0.0.1:5505/api`. Запрос `getTasks({ completed: false, priority: "high" })`. Фильтры передаются серверу в query.

### Листинг 3. Клиент, getTasks и вызов

```text
const baseUrl = process.env.API_URL ?? "http://127.0.0.1:5505/api";
const client = createApiClient({ baseUrl, timeoutMs: 2000 });
const api = createTaskApi(client);

async function getTasks(filters = {}) {
  const query = normalizeFilters(filters);
  const payload = await client.requestJson("/tasks", { query });
  const data = assertCollection(payload, "GET /tasks");
  data.forEach((task) => assertTask(task, "GET /tasks"));
  const meta = assertMeta(payload, data, "GET /tasks");
  return { tasks: data, meta };
}

const { tasks, meta } = await api.getTasks({
  completed: false,
  priority: "high",
});
```

`normalizeFilters` кладёт `completed` и `priority` в объект `query`. `requestJson` собирает URL через `URLSearchParams`, поэтому запрос уходит как `GET /api/tasks?completed=false&priority=high`.

## 4. Обработка HTTP-ошибки — 1 балл

API ПР5 оставлен запущенным. Вызов `getTaskById(999)` внутри `try/catch`.

### Листинг 4. Проверка статуса и перехват

```text
if (!response.ok) {
  throw new ApiError(
    "http",
    apiError?.message ?? `Сервер ответил статусом ${response.status}.`,
    { url, status: response.status, code: apiError?.code, details: apiError?.details }
  );
}

try {
  await api.getTaskById(999);
} catch (error) {
  console.log(error.kind, error.status);
}
```

`fetch` считает ответ 404 успешно доставленным `Response`. Без проверки `response.ok` клиент пошёл бы разбирать тело как задачу. Явная проверка переводит статус в `ApiError` с `kind: "http"`, и `await` в `try/catch` не оставляет необработанного исключения.

### Результат прогона

```text
перехвачено: kind=http status=404 code=TASK_NOT_FOUND | Задача с id 999 не найдена.
details: {"id":999}
```

Скриншот 4. Вставить снимок с `kind` и `status`.

Пояснение. Клиент явно проверяет HTTP-статус, потому что `fetch` не считает 404 отказом сети: без `response.ok` ошибка «задача не найдена» смешалась бы с успешным JSON.

### Листинг 5. Смена статуса, PATCH и повторная загрузка

```text
async function toggleTask(id) {
  const task = tasks.find((item) => item.id === id);
  await mutate(
    () => api.setTaskCompleted(id, !task.completed),
    `Статус задачи ${id} изменён.`
  );
}

async function setTaskCompleted(id, completed) {
  return updateTask(id, { completed });
}

async function updateTask(id, changes) {
  const payload = await client.requestJson(`/tasks/${id}`, {
    method: "PATCH",
    body: changes,
  });
  return unwrapTask(payload, "PATCH /tasks/:id", id);
}

async function mutate(action, successMessage) {
  await action();
  await loadCurrentTasks();
}
```

`loadCurrentTasks` снова вызывает `getTasks` с текущими фильтрами. При `completed=false` выполненная задача 4 в ответ `GET` уже не входит, поэтому карточка пропадает.
